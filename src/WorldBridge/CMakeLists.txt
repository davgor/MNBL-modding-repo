cmake_minimum_required(VERSION 3.20)
project(MNBLWorldBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # /Brepro: deterministic PE timestamps so identical sources give identical binaries.
  add_compile_options(/W4 /WX /permissive- /fp:precise /utf-8 /Brepro)
  add_link_options(/Brepro)
else()
  add_compile_options(-Wall -Wextra -Werror -ffp-contract=off)
endif()
add_library(worldbridge_process STATIC src/producer_process.cpp)
target_include_directories(worldbridge_process PUBLIC include)
target_compile_definitions(worldbridge_process PRIVATE WORLDBRIDGE_CMAKE_VERSION="${CMAKE_VERSION}")
add_library(worldbridge_source STATIC src/json_stream.cpp src/source_reader.cpp src/canonical_json.cpp src/world_identity.cpp src/attempt_schedule.cpp)
target_include_directories(worldbridge_source PUBLIC include)
add_library(worldbridge_projection STATIC src/source_nodes.cpp src/projection.cpp src/projected_world.cpp src/material_compiler.cpp src/terrain_compiler.cpp src/terrain_layers.cpp src/terrain_shaping.cpp src/stable_ids.cpp src/campaign_compiler.cpp src/travel_graph.cpp src/port_compiler.cpp src/village_quota.cpp src/name_cleaner.cpp src/population.cpp src/bundle_writer.cpp src/bundle_validator.cpp src/policy_map2.cpp commands/inspect_bundle.cpp)
target_compile_definitions(worldbridge_projection PRIVATE WORLDBRIDGE_VERSION="${PROJECT_VERSION}")
target_include_directories(worldbridge_projection PUBLIC include)
target_link_libraries(worldbridge_projection PUBLIC worldbridge_source)
add_executable(worldbridge main.cpp surface.cpp src/generation_driver.cpp)
target_link_libraries(worldbridge PRIVATE worldbridge_process worldbridge_source worldbridge_projection)
add_executable(fake_producer tests/fake_producer.cpp)
add_executable(process_contract_tests tests/process_contract_tests.cpp)
target_link_libraries(process_contract_tests PRIVATE worldbridge_process)
add_executable(source_reader_tests tests/source_reader_tests.cpp)
target_link_libraries(source_reader_tests PRIVATE worldbridge_source)
add_executable(identity_retry_tests tests/identity_retry_tests.cpp)
target_link_libraries(identity_retry_tests PRIVATE worldbridge_source)
add_executable(projection_tests tests/projection_tests.cpp)
target_link_libraries(projection_tests PRIVATE worldbridge_projection)
add_executable(terrain_resampling_tests tests/terrain_resampling_tests.cpp)
target_link_libraries(terrain_resampling_tests PRIVATE worldbridge_projection)
add_executable(terrain_shaping_tests tests/terrain_shaping_tests.cpp)
target_link_libraries(terrain_shaping_tests PRIVATE worldbridge_projection)
add_executable(campaign_semantics_tests tests/campaign_semantics_tests.cpp)
target_link_libraries(campaign_semantics_tests PRIVATE worldbridge_projection)
add_executable(travel_graph_tests tests/travel_graph_tests.cpp)
target_link_libraries(travel_graph_tests PRIVATE worldbridge_projection)
add_executable(bundle_transaction_tests tests/bundle_transaction_tests.cpp)
target_link_libraries(bundle_transaction_tests PRIVATE worldbridge_projection)
add_executable(map2_policy_tests tests/map2_policy_tests.cpp)
target_link_libraries(map2_policy_tests PRIVATE worldbridge_projection)
add_executable(population_tests tests/population_tests.cpp)
target_link_libraries(population_tests PRIVATE worldbridge_projection)
enable_testing()
add_test(NAME reader_contract COMMAND worldbridge --self-test)
add_test(NAME process_contract COMMAND process_contract_tests $<TARGET_FILE:fake_producer> ${CMAKE_CURRENT_BINARY_DIR}/process-contract)
add_test(NAME identity_retry COMMAND identity_retry_tests ${CMAKE_CURRENT_BINARY_DIR}/identity-retry)
add_test(NAME source_contract COMMAND source_reader_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/source_contract ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/contracts/source-compatibility.json ${CMAKE_CURRENT_BINARY_DIR}/source-contract)
add_test(NAME projection COMMAND projection_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/projection.json ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/calibration/axis-sea-datum/calibration.json ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/source_contract ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/contracts/source-compatibility.json ${CMAKE_CURRENT_BINARY_DIR}/projection)
add_test(NAME terrain COMMAND terrain_resampling_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/projection.json ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/terrain-materials.json ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/calibration/terrain-import/terrain-import.json ${CMAKE_CURRENT_BINARY_DIR}/terrain)
add_test(NAME shaping COMMAND terrain_shaping_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/projection.json ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/terrain-materials.json ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/calibration/terrain-import/terrain-import.json ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/terrain-shaping.json ${CMAKE_CURRENT_BINARY_DIR}/shaping)
add_test(NAME campaign COMMAND campaign_semantics_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/campaign-content.json ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/culture-presentations.json ${CMAKE_CURRENT_BINARY_DIR}/campaign)
add_test(NAME travel COMMAND travel_graph_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies/playability.json ${CMAKE_CURRENT_BINARY_DIR}/travel)
add_test(NAME bundle COMMAND bundle_transaction_tests ${CMAKE_CURRENT_BINARY_DIR}/bundle)
add_test(NAME population COMMAND population_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies)
add_test(NAME map2_policies COMMAND map2_policy_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../worldgen/policies ${CMAKE_CURRENT_BINARY_DIR}/map2-policies)
